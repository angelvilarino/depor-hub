# L-19: carrera en syncReminders

Fecha: 2026-10-31T11:41:26Z

## Efecto sin serialización
import { useEffect } from 'react';
import { useFixtures } from '@/data/queries';
import { useSettings } from '@/store/settings';
import { syncReminders } from './notifications';

/** Sin UI: reprograma los recordatorios cada vez que cambian los datos o los ajustes. */
export function ReminderSync() {
  const fixtures = useFixtures().data;
  const { remindersEnabled, remindMatches, language } = useSettings();

  useEffect(() => {
    if (!fixtures) return;
    syncReminders(fixtures.matches, remindersEnabled ? remindMatches : {}).catch(() => {
      // Sin permisos o sin soporte (p. ej. web): no es crítico.
    });
  }, [fixtures, remindersEnabled, remindMatches, language]);

  return null;
}

## Función: cancela todo y reprograma en pasos async
export async function syncReminders(matches: MatchT[], remindMatches: ReminderMap): Promise<void> {
  const scheduled = await Notifications.getAllScheduledNotificationsAsync();
  await Promise.all(
    scheduled
      .filter((n) => n.content.data?.kind === KIND)
      .map((n) => Notifications.cancelScheduledNotificationAsync(n.identifier)),
  );
  const plan = planReminders(matches, remindMatches, Date.now());
  if (plan.length === 0) return;
  await ensureChannel();
  const locale = i18n.language;
  const name = (t: { name: string; shortName: string | null }) => t.shortName ?? t.name;
  for (const p of plan) {
    const time = formatMatchTime(p.match.utcDate, p.match.status, locale) ?? '';
    await Notifications.scheduleNotificationAsync({
      identifier: p.id,
      content: {
        title: `${name(p.match.home)} – ${name(p.match.away)}`,
        body: i18n.t(`reminder.body.${offsetKey(p.offsetMin)}`, { time }),
        data: { kind: KIND, matchId: p.match.id },
      },
      trigger: { type: Notifications.SchedulableTriggerInputTypes.DATE, date: p.fireAt, channelId: CHANNEL },
    });
  }
}

## Escenario
Dos cambios seguidos de fixtures/remindMatches (p. ej. revalidación al volver a primer plano) lanzan dos syncReminders solapados. A puede cancelar lo que B acaba de programar y dejar el plan de A o una mezcla. No hay mutex, contador de generación ni cancelación.

Confianza: Probable (código); no reproducido en dispositivo.
