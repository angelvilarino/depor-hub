        // news.json aún no existe (H6): 404 = «sin titulares», no un error global.

      }
    },
    staleTime: __DEV__ ? 0 : staleTime,
    gcTime: GC_TIME,
    retry: (count, err) =>
      !(
        err instanceof DataError &&
        (err.kind === 'not_configured' || err.kind === 'invalid' || err.status === 404)
      ) && count < 2,
  });
